const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); const cors = require('cors'); const app = express(); app.use(cors()); const server = http.createServer(app); const io = new Server(server, { cors: { origin: "*", methods: ["GET", "POST"] } }); io.on('connection', (socket) => { console.log('Bir kullanıcı bağlandı:', socket.id); // Lobi mesajlarını dinle ve herkese ilet socket.on('chat-message', (data) => { io.emit('chat-message', data); }); socket.on('disconnect', () => { console.log('Kullanıcı ayrıldı:', socket.id); }); }); const PORT = process.env.PORT || 3000; server.listen(PORT, () => { console.log(`Sunucu ${PORT} portunda aktif!`); });
